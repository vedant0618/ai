import heapq

goal = (1, 2, 3, 4, 5, 6, 7, 8, 0)
start = (1, 2, 3, 4, 0, 6, 7, 5, 8)


def heuristic(state):
  distance = 0
  for value in range(1, 9):
    current_index = state.index(value)
    goal_index = goal.index(value)
    current_row, current_col = divmod(current_index, 3)
    goal_row, goal_col = divmod(goal_index, 3)
    distance += abs(current_row - goal_row) + abs(current_col - goal_col)
  return distance


def neighbours(state):
  result = []
  blank = state.index(0)
  row, col = divmod(blank, 3)
  moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]  # Fixed typo here
  for dr, dc in moves:
    nr = row + dr
    nc = col + dc
    if 0 <= nr < 3 and 0 <= nc < 3:
      new_blank = nr * 3 + nc
      new_state = list(state)
      new_state[blank], new_state[new_blank] = (
          new_state[new_blank],
          new_state[blank],
      )
      result.append(tuple(new_state))
  return result


open_list = []
heapq.heappush(open_list, (heuristic(start), 0, start, [start]))
visited = set()

while open_list:
  f, g, current, path = heapq.heappop(open_list)
  print("--------------------------------")
  print("Current State")
  print(current[0:3])
  print(current[3:6])
  print(current[6:9])
  print("g =", g)
  print("h =", heuristic(current))
  print("f =", f)

  if current == goal:
    print("\nGoal Found!")
    print("Total Moves =", g)
    print("\nSolution Path")
    for step in path:
      print(step[0:3])
      print(step[3:6])
      print(step[6:9])
      print()
    break

  if current in visited:
    continue
  visited.add(current)

  for nxt in neighbours(current):
    if nxt not in visited:
      new_g = g + 1
      new_h = heuristic(nxt)
      new_f = new_g + new_h
      heapq.heappush(open_list, (new_f, new_g, nxt, path + [nxt]))

print("Search Completed.")
